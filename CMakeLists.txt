cmake_minimum_required(VERSION 3.15)
project(OptimalAircraftScheduling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(flight_scheduler src/flight_scheduler.cpp)
if(MSVC)
  target_compile_options(flight_scheduler PRIVATE /W4)
else()
  target_compile_options(flight_scheduler PRIVATE -Wall -Wextra -Wpedantic)
endif()
